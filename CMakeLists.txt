# dasImguiImplot builds two ways from a single CMakeLists:
#  - included from a daslang superbuild (the ADD_MODULE_* macros exist): the module
#    links statically into the host app via the convention path (dasImgui provides
#    imgui headers + the libDasModuleImgui target) -- how an embedding app and the
#    fat-man static-link example consume it.
#  - configured directly (top-level project, DASLANG_DIR set): the standalone
#    .shared_module build CI, the daspkg index, and -load_module consumers run on.
IF(NOT COMMAND ADD_MODULE_CPP)
    cmake_minimum_required(VERSION 3.16)
    project(dasImguiImplot CXX C)
ENDIF()

IF((NOT DAS_IMPLOT_INCLUDED) AND ((NOT ${DAS_IMPLOT_DISABLED}) OR (NOT DEFINED DAS_IMPLOT_DISABLED)))
SET(DAS_IMPLOT_INCLUDED TRUE)
MESSAGE(STATUS "dasImguiImplot module included.")

SET(DAS_IMPLOT_DIR ${CMAKE_CURRENT_LIST_DIR})

IF(COMMAND ADD_MODULE_CPP)
    # superbuild: dasImgui's CMakeLists (included earlier -- the modules glob is
    # alphabetical) already set imgui_INCLUDE_DIR and created libDasModuleImgui
    IF(NOT DEFINED imgui_INCLUDE_DIR)
        MESSAGE(FATAL_ERROR "dasImguiImplot superbuild: dasImgui must be included first (imgui_INCLUDE_DIR unset)")
    ENDIF()
    SET(DAS_IMGUI_DIR ${CMAKE_CURRENT_LIST_DIR}/../dasImgui)
ELSE()
    # DASLANG_DIR must point to the daslang SDK root
    if(NOT DEFINED DASLANG_DIR)
        message(FATAL_ERROR "DASLANG_DIR must be set to the daslang SDK root")
    endif()

# Locate dasImgui (sibling module under same modules/ parent)
get_filename_component(MODULES_DIR "${DAS_IMPLOT_DIR}" DIRECTORY)
SET(DAS_IMGUI_DIR "${MODULES_DIR}/dasImgui")
if(NOT EXISTS "${DAS_IMGUI_DIR}/src/cb_dasIMGUI.h")
    message(FATAL_ERROR "dasImgui not found at ${DAS_IMGUI_DIR}. dasImgui is part of the daslang tree (modules/dasImgui) — build daslang first, or point DASLANG_DIR at a daslang tree and install this package under its modules/ (daspkg install).")
endif()

# imgui headers (from dasImgui's FetchContent download)
if(EXISTS "${DAS_IMGUI_DIR}/imgui/imgui.h")
    SET(imgui_INCLUDE_DIR "${DAS_IMGUI_DIR}/imgui")
else()
    message(FATAL_ERROR "imgui headers not found at ${DAS_IMGUI_DIR}/imgui/. Build dasImgui first.")
endif()

# dasModuleImgui import library (from dasImgui build)
if(WIN32)
    SET(DAS_MODULE_IMGUI_LIB "${DAS_IMGUI_DIR}/dasModuleImgui.lib")
    if(NOT EXISTS "${DAS_MODULE_IMGUI_LIB}")
        message(FATAL_ERROR "dasModuleImgui.lib not found at ${DAS_IMGUI_DIR}/. Build dasImgui first.")
    endif()
endif()
ENDIF()

# ImPlot v0.16 (vendored, patched for imgui 1.90.6 compatibility)
SET(IMPLOT_INCLUDE_DIR ${DAS_IMPLOT_DIR}/implot)

# ImPlot source files (implot_demo.cpp intentionally excluded — reference only)
SET(DAS_IMPLOT_SRC
    ${IMPLOT_INCLUDE_DIR}/implot.cpp
    ${IMPLOT_INCLUDE_DIR}/implot_items.cpp
)

# binding source files (generated by bind/bind_implot.das + hand-written main.cpp).
# The func_*.cpp chunk count varies with the bound-function count, so glob them — the
# generator can produce fewer/more chunks across regens. Always `rm src/dasIMPLOT.func_*.cpp`
# before regenerating (stale higher-numbered chunks otherwise linger and get compiled),
# then re-run CMake configure so this glob re-evaluates.
file(GLOB DAS_IMPLOT_FUNC_SRC "${DAS_IMPLOT_DIR}/src/dasIMPLOT.func_*.cpp")
SET(DAS_IMPLOT_BIND_SRC
    ${DAS_IMPLOT_DIR}/src/dasIMPLOT.cpp
    ${DAS_IMPLOT_DIR}/src/dasIMPLOT.main.cpp
    ${DAS_IMPLOT_FUNC_SRC}
)


IF(COMMAND ADD_MODULE_CPP)
    # ===== daslang superbuild: static-embed =====
    # `require imgui/<name>` -> modules/dasImguiImplot/daslib/<name>.das; the glob IS the
    # .das_module register_native_path list (file stem == module name throughout).
    ADD_MODULE_CPP(dasIMPLOT)
    FILE(GLOB _dasImguiImplot_das_files "${DAS_IMPLOT_DIR}/daslib/*.das")
    FOREACH(_f ${_dasImguiImplot_das_files})
        GET_FILENAME_COMPONENT(_n "${_f}" NAME_WE)
        ADD_MODULE_DAS(imgui daslib ${_n})
    ENDFOREACH()

    # static half only -- no ADD_MODULE_LIB shared twin: in a superbuild the DLL
    # flavor comes from this module's own standalone build, and the twin would need
    # the dllexport/dllimport matrix against dasImgui for nothing a host ever loads
    ADD_DAS_STATIC_MODULE_LIB(libDasModuleImplot ${DAS_IMPLOT_BIND_SRC} ${DAS_IMPLOT_SRC})
    SET(_dasImguiImplot_targets libDasModuleImplot)
    FOREACH(_t ${_dasImguiImplot_targets})
        TARGET_INCLUDE_DIRECTORIES(${_t} PRIVATE
            ${imgui_INCLUDE_DIR}
            ${IMPLOT_INCLUDE_DIR}
            ${DAS_IMGUI_DIR}/src
            ${DAS_IMPLOT_DIR}/src
        )
        TARGET_COMPILE_DEFINITIONS(${_t} PRIVATE IMGUI_DISABLE_OBSOLETE_FUNCTIONS=1)
        SETUP_CPP11(${_t})
    ENDFOREACH()
    # imgui core symbols come from dasImgui's libs
    TARGET_LINK_LIBRARIES(libDasModuleImplot PUBLIC libDasModuleImgui)
ELSE()
IF(WIN32)
    SET(IMGUI_IMPORT_ATTR "__declspec\(dllimport\)")
ELSE()
    SET(IMGUI_IMPORT_ATTR "")
ENDIF()

# dasModuleImplot.shared_module
add_library(dasModuleImplot SHARED ${DAS_IMPLOT_BIND_SRC} ${DAS_IMPLOT_SRC})
target_compile_features(dasModuleImplot PRIVATE cxx_std_17)
target_compile_definitions(dasModuleImplot PRIVATE
    DAS_MOD_EXPORTS DAS_ENABLE_DLL=1
    IMGUI_DISABLE_OBSOLETE_FUNCTIONS=1
    IMPLOT_DISABLE_OBSOLETE_FUNCTIONS=1
    IMGUI_API=${IMGUI_IMPORT_ATTR}
    # Must match dasImgui's ImWchar width, or ImWchar-taking imgui symbols mangle as
    # unsigned short here and unsigned int there, and the link fails on FindGlyph.
    IMGUI_USE_WCHAR32=1
    $<$<CONFIG:Release>:DAS_NO_ASSERTIONS>
    $<$<CONFIG:MinSizeRel>:DAS_NO_ASSERTIONS>
    $<$<CONFIG:RelWithDebInfo>:DAS_NO_ASSERTIONS>
)
# Trim ImPlot's templated Plot* instantiations to the types we bind (float/double/int);
# the default list instantiates all 10 numeric widths. Matches the hand-written
# forwarders in src/dasIMPLOT.main.cpp.
target_compile_definitions(dasModuleImplot PRIVATE "IMPLOT_CUSTOM_NUMERIC_TYPES=(float)(double)(ImS32)")
target_include_directories(dasModuleImplot PRIVATE
    ${DASLANG_DIR}/include
    ${imgui_INCLUDE_DIR}
    ${IMPLOT_INCLUDE_DIR}
    ${DAS_IMGUI_DIR}/src
    ${DAS_IMPLOT_DIR}/src
)
if(WIN32)
    if(EXISTS "${DASLANG_DIR}/lib/Release/libDaScriptDyn.lib")
        target_link_libraries(dasModuleImplot PRIVATE
            ${DASLANG_DIR}/lib/Release/libDaScriptDyn.lib
            ${DASLANG_DIR}/lib/Release/libDaScriptDyn_runtime.lib
            ${DAS_MODULE_IMGUI_LIB}
        )
    else()
        target_link_libraries(dasModuleImplot PRIVATE
            ${DASLANG_DIR}/lib/libDaScriptDyn.lib
            ${DASLANG_DIR}/lib/libDaScriptDyn_runtime.lib
            ${DAS_MODULE_IMGUI_LIB}
        )
    endif()
elseif(APPLE)
    target_link_libraries(dasModuleImplot PRIVATE
        ${DASLANG_DIR}/lib/liblibDaScriptDyn.dylib
        ${DASLANG_DIR}/lib/liblibDaScriptDyn_runtime.dylib
        ${DAS_IMGUI_DIR}/dasModuleImgui.shared_module
    )
else()
    target_link_libraries(dasModuleImplot PRIVATE
        ${DASLANG_DIR}/lib/liblibDaScriptDyn.so
        ${DASLANG_DIR}/lib/liblibDaScriptDyn_runtime.so
        ${DAS_IMGUI_DIR}/dasModuleImgui.shared_module
    )
endif()
if(MSVC)
    target_compile_options(dasModuleImplot PRIVATE /wd4005 /bigobj)
else()
    # daslang runtime libs are built -fno-rtti for optimized configs; match the ABI
    # (mirror dasImgui's SETUP_OPTIONS) or the .shared_module mislinks.
    target_compile_options(dasModuleImplot PRIVATE
        $<$<CONFIG:Release>:-fno-rtti>
        $<$<CONFIG:MinSizeRel>:-fno-rtti>
        $<$<CONFIG:RelWithDebInfo>:-fno-rtti>)
endif()
set_target_properties(dasModuleImplot PROPERTIES
    SUFFIX ".shared_module"
    PREFIX ""
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${DAS_IMPLOT_DIR}"
    LIBRARY_OUTPUT_DIRECTORY "${DAS_IMPLOT_DIR}"
)
ENDIF() # superbuild / standalone

ENDIF() # DAS_IMPLOT_INCLUDED guard
